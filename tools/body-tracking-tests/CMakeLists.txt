cmake_minimum_required(VERSION 3.20)
project(BodyTrackingTests LANGUAGES CXX)
get_filename_component(MCOSU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MCENGINE_ROOT "${MCOSU_ROOT}/third_party/McEngine/McEngine" CACHE PATH "McEngine source directory")
add_executable(body-tracking-tests main.cpp "${MCOSU_ROOT}/src/App/Osu/Avatar/OpenVRBodyTracking.cpp" "${MCENGINE_ROOT}/src/Util/Matrices.cpp")
target_compile_features(body-tracking-tests PRIVATE cxx_std_11)
target_include_directories(body-tracking-tests PRIVATE "${MCOSU_ROOT}/src/App/Osu/Avatar" "${MCOSU_ROOT}/third_party/openvr-input" "${MCENGINE_ROOT}/src/Util")
target_link_libraries(body-tracking-tests PRIVATE "${MCENGINE_ROOT}/libraries/openvr/lib/windows/openvr_api.lib")
enable_testing()
add_test(NAME body-tracking-assignment COMMAND body-tracking-tests)
